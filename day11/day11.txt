

Joins in MySQL
Joins in MySQL are used to retrieve data from multiple tables 
based on a related column.
1. INNER JOIN
	Explanation:
The INNER JOIN keyword selects records that have matching values in both tables.
If there is no match, the rows are excluded from the result set.
This is the most commonly used join in SQL.
It is useful when we only need records
 that have corresponding entries in both tables.

syntax
select * from table1 t1 inner join table2 t2 
                                on t1.refcol=t2.refcol;

ex
select * from emp e  join dept d on e.deptno=d.deptno;


get the empname and their dnames only 
select e.ename,d.dname from emp e  join
                    dept d on e.deptno=d.deptno;

get the details who are working in newyork
select * from emp e join dept d on
              e.deptno=d.deptno where d.loc='new york';














Queries on joins
1) Display student_id who registered under anitha
select * from trainers t join enrollments e on 
   t.trainer_id=e.trainer_id where t.trainer_name='anita';

2) Display student_id who registered for python
select * from courses c join enrollments e on
     c.course_id=e.course_id where c.course_name='python';

3) Display who was teaching the course having highest fees
select * from trainers t join courses c on 
     t.expertise=c.course_name order by c.fee desc limit 1;

4) Display sid,cid,cname,fees of each std
select e.student_id,c.course_id,c.course_name,c.fee from 
    courses c join enrollments e on c.course_id=e.course_id;
    
5) Display studentid,total fees paid by each student
select e.student_id,sum(c.fee) from courses c join enrollments e
  on c.course_id=e.course_id group by e.student_id;

6) Display coursename,total fees accumulate in  each course
select c.course_name,sum(c.fee) from courses c join enrollments e
  on c.course_id=e.course_id group by c.course_name;

7) Display sid,trianername, subject,fee for all students
select e.student_id ,t.trainer_name, t.expertise, c.fee from
    enrollments e join trainers t on  e.trainer_id=t.trainer_id 
    join courses c  on  c.course_id=e.course_id;

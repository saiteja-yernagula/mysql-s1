Subqueries
1) Employees who work in the Sales department
select ename from emp where deptno=(select
    deptno from dept where dname='sales');

2) Employees earning more than the average salary
select ename from emp sal> (select avg(sal) from emp);

3) Employees who work in the same department as SMITH
select ename from emp where deptno=(select 
    deptno from emp where ename='smith');

4) Highest paid employee
select ename from emp where sal=(select max(sal) from emp);

5) Employees who work in NEW YORK
select ename from emp where deptno=
     (select deptno from dept where loc='new york');

6) Employees whose salary is greater than FORD’s salary
select ename from emp where sal > 
     (select sal from emp where ename='ford');


1) Students who enrolled in Python by course id
select student_name from students where student_id
  in (select student_id from enrollments where course_id=101);

2) Students who enrolled in any course
select student_name from students where student_id 
    in (select student_id from enrollments);

3) Students who did NOT enroll anywhere
select student_name from students where student_id 
   not in (select student_id from enrollments);

4) Courses that cost more than average fee
select course_name from courses where fee 
    >(select avg(fee) from courses);


6) Show which course have highest course fee

7) Show which course have lowest course fee
select * from courses where fee=(select min(fee) from courses);

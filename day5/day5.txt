operators

arithmetic operators
+  select  sal+1000 from emp;
-  select  sal-1000 from emp;
* select ename,sal*12 as annual_salary from emp;
/ select ename,sal/2 as halfmonth_salary from emp;
% select 10%2 from dual;

comparsion operators
= 
  select * from emp where job='manager';
  select * from emp where deptno=10;
  select * from emp where empno=7782;
<> or !=
  select * from emp where deptno<>10;
   select * from emp where job!='salesman';
<
    select * from emp where sal<2000;
>
    select * from emp where sal>2000;
<=
select * from emp where deptno<=20;
>=
select * from emp where deptno>=20;

logical operators
and
 select * from emp where deptno=20 and job='clerk';
 select * from emp where sal<2000 and deptno=20;
or 
  select * from emp where sal<2000 or deptno=20;
  select * from emp where job='salesman' or deptno=30;
not
select * from emp where not job='manager';
 select * from emp where not deptno=10;
 select * from emp where not job='salesman' 
                        and not deptno=30;

special operators
in- it is used to give the records which are in 
    a set of values
 select * from emp where deptno in (10,20,50);

not in
 select * from emp where deptno not in (10,20,50);
 select * from emp where job not in ("salesman","clerk");

is NULL- it checks whether a value is NULL
 select * from emp where comm is null;
  select * from emp where mgr is null;

is not null - 
 select * from emp where mgr is not null;
  select * from emp where comm is not null;

between - it will check whether the records
   are in a range
 select * from emp where sal between 1000 and 2000;
not between- it will give the values which are 
out of range
select * from emp where sal not between 1000 and 2000;


wildcards
it is used to match a specific pattern 
using like operator

% - for matching multiple characters
 select * from emp where ename like "s%";//starting with
 select * from emp where ename like "%s";//ends with
 select * from emp where ename like "%i%";//if it contains

_ - for matching single characters
 select * from emp where ename like "__rd";
 select * from emp where ename like "__r%";
